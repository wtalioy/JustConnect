cmake_minimum_required(VERSION 3.20)
project(JustConnectSystemNetworkHelper VERSION 0.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The system network helper package is Linux-only")
endif()
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
# This build deliberately selects the distribution SDK, independently of the GUI's Qt SDK.
find_package(Qt6 6.10 REQUIRED COMPONENTS Core)
include("${REPO}/cmake/SourceRevision.cmake")
include("${REPO}/cmake/NetworkHelper.cmake")
justconnect_linux_network_helper()
include(CTest)
if(BUILD_TESTING)
    justconnect_linux_network_helper_tests()
endif()

install(FILES "${CMAKE_CURRENT_BINARY_DIR}/source-revision.txt" "${REPO}/LICENSE"
    DESTINATION share/doc/justconnect-network-helper COMPONENT LinuxNetworkHelper)

set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME justconnect-network-helper)
set(CPACK_PACKAGE_VENDOR JustConnect)
set(CPACK_PACKAGE_CONTACT "JustConnect maintainers (https://github.com/wtalioy/JustConnect/issues)")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/wtalioy/JustConnect")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Minimal privileged device and route helper for JustConnect")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_PACKAGE_VERSION "0~git${REVISION_TIME}.${REVISION}")
set(CPACK_DEBIAN_PACKAGE_SECTION net)
set(CPACK_DEBIAN_PACKAGE_PRIORITY optional)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "iproute2, pkexec | policykit-1")
# dpkg-shlibdeps records actual distribution Qt/libc/C++ requirements, including minimum versions.
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_COMPRESSION_TYPE xz)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
include(CPack)
