cmake_minimum_required(VERSION 3.20)
project(JustConnectWindowsNetworkHelper LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The Windows system helper requires Windows")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
include("${REPO}/cmake/WindowsNetworkHelper.cmake")
justconnect_windows_network_helper(windows_helper)
file(READ "${REPO}/resource/network-helper-windows.json" LAYOUT)
foreach(FIELD IN ITEMS driver_library driver_license)
    string(JSON NAME GET "${LAYOUT}" ${FIELD})
    install(FILES "${WINTUN_INCLUDE_DIR}/${NAME}" DESTINATION . COMPONENT WindowsNetworkHelper)
endforeach()
install(TARGETS windows_helper RUNTIME DESTINATION . COMPONENT WindowsNetworkHelper)
