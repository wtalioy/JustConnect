cmake_minimum_required(VERSION 3.20)
project(JustConnectWindowsDevice LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The device fixture is Windows-only")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${REPO}/cmake/WindowsNetworkHelper.cmake")
include("${REPO}/cmake/SystemIntegration.cmake")
include("${REPO}/cmake/SourceRevision.cmake")
justconnect_windows_dns_reader(JustConnectDnsReader)
justconnect_windows_dns_reader(windows_dnsreader_test)
target_compile_definitions(windows_dnsreader_test PRIVATE JC_DNS_READER_TESTING)
justconnect_windows_network_helper(windows_helper)
get_target_property(HELPER_OUTPUT windows_helper OUTPUT_NAME)
add_executable(windows_helper_previous "${REPO}/infrastructure/networkhelper/windows/main.cpp")
add_executable(windows_helper_startup_failure servicefailure.cpp)
foreach(VARIANT IN ITEMS previous startup_failure)
    target_link_libraries(windows_helper_${VARIANT} PRIVATE windows_helper_support)
    target_compile_definitions(windows_helper_${VARIANT} PRIVATE JC_HELPER_REVISION="${REVISION}-${VARIANT}" JC_HELPER_SOURCE_STATE="${SOURCE_STATE}")
    set_target_properties(windows_helper_${VARIANT} PROPERTIES OUTPUT_NAME "${HELPER_OUTPUT}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${VARIANT}")
endforeach()
add_library(windows_fixture_support STATIC ordinarytoken.cpp)
target_link_libraries(windows_fixture_support PUBLIC windows_helper_support)
add_executable(windows_device_broker broker.cpp)
target_link_libraries(windows_device_broker PRIVATE windows_fixture_support setupapi uuid psapi)

foreach(CASE IN ITEMS process_identity registry_journal packet_frame nrpt installation recovery system_dns control session)
    add_executable(windows_${CASE}_test ${CASE}.cpp)
    target_link_libraries(windows_${CASE}_test PRIVATE windows_helper_support windows_fixture_support)
endforeach()

option(JC_WINDOWS_QT_CONTROL_TEST "Build the ordinary Qt ownership control client checks" OFF)
if(JC_WINDOWS_QT_CONTROL_TEST)
    justconnect_windows_network_client(windows_control_client)
    add_executable(windows_control_qt_test qt_control.cpp)
    set_target_properties(windows_control_qt_test PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    target_link_libraries(windows_control_qt_test PRIVATE windows_control_client ws2_32 iphlpapi)
endif()
