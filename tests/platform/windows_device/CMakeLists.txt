cmake_minimum_required(VERSION 3.20)
project(JustConnectWindowsDevice LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The device fixture is Windows-only")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${REPO}/cmake/WindowsNetworkHelper.cmake")
justconnect_windows_network_helper(windows_helper)
add_library(windows_fixture_support STATIC ordinarytoken.cpp)
target_link_libraries(windows_fixture_support PUBLIC windows_helper_support)
add_executable(windows_device_broker broker.cpp)
target_link_libraries(windows_device_broker PRIVATE windows_fixture_support setupapi uuid psapi)

foreach(CASE IN ITEMS process_identity publisher registry_journal packet_frame nrpt installation recovery system_dns control session)
    add_executable(windows_${CASE}_test ${CASE}.cpp)
    target_link_libraries(windows_${CASE}_test PRIVATE windows_helper_support windows_fixture_support)
endforeach()

option(JC_WINDOWS_QT_CONTROL_TEST "Build the ordinary Qt ownership control client checks" OFF)
if(JC_WINDOWS_QT_CONTROL_TEST)
    justconnect_windows_network_client(windows_control_client)
    add_executable(windows_control_qt_test qt_control.cpp)
    set_target_properties(windows_control_qt_test PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    target_link_libraries(windows_control_qt_test PRIVATE windows_control_client ws2_32 iphlpapi)
endif()
